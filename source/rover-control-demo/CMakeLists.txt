cmake_minimum_required(VERSION 3.20)
project(RoverControlDemo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
add_executable(rover_demo src/demo.cpp)
target_include_directories(rover_demo PRIVATE include)
if(BUILD_TESTING)
    foreach(name drive_logic motor_control_core remote_command)
        add_executable(${name}_test tests/${name}_test.cpp)
        target_include_directories(${name}_test PRIVATE include)
        add_test(NAME ${name} COMMAND ${name}_test)
    endforeach()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME demo COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/demo_test.py $<TARGET_FILE:rover_demo>)
endif()
