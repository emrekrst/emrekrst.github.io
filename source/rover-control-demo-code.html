# Rover command, ramp and watchdog demo

A small hardware-free extraction from an internship rover control project. It preserves the original C++ command parser/sequence gate, keyboard mixer, acceleration/deceleration ramp, and normalized differential-drive controller with a command watchdog. It does not contain ROS, serial drivers, motor control transports or deployment configuration.

## Run

Requires a C++17 compiler, CMake 3.20+ and Python 3 for the CLI test.

```bash
cmake -S . -B build -DCMAKE_BUILD_TYPE=Release
cmake --build build --parallel
ctest --test-dir build --output-on-failure
./build/rover_demo
```

The CLI uses deterministic simulated time and prints CSV. It accepts a `ROVER1` command at 0 ms, rejects its duplicate at 20 ms, accepts the next sequence at 40 ms, rejects a malformed packet at 60 ms, stops at 140 ms when command age reaches 100 ms, and resumes from zero after a fresh command at 180 ms. See [samples/demo.csv](samples/demo.csv).

`include/drive_logic.hpp`, `include/motor_teleop_v2/motor_control_core.hpp`, `include/remote_mvp/remote_command.hpp` and their original tests were extracted without modification. `src/demo.cpp` and its subprocess test are a new demonstration harness. The sequence gate deduplicates within the current session; it is not an authentication or general replay-protection mechanism. This CLI never binds a socket or opens a serial port.

## Evidence and rights

See [VERIFICATION.md](VERIFICATION.md) for results and limits. This sample supports the command/ramp/watchdog aspect of the project. It does not reproduce Jetson deployment, F310 input, RealSense video, ROS 2 telemetry, BusLinker/servo communication, or the historical UART baud diagnosis.

Prepared from Emre Karausta's internship working copy, retaining original source comments. No source license was present in this extraction and no blanket license is inferred. This is a local release candidate; nothing has been published.

#pragma once

#include <stdexcept>
#include <algorithm>
#include <cmath>

struct KeyState {
  bool w{};
  bool a{};
  bool s{};
  bool d{};
};

struct WheelCommand {
  double left{};
  double right{};
};

inline WheelCommand mixKeys(const KeyState& keys, double inner_ratio) {
  if (!std::isfinite(inner_ratio) || inner_ratio < 0.0 || inner_ratio > 1.0) {
    throw std::invalid_argument("inner ratio must be between zero and one");
  }
  const int translation = static_cast<int>(keys.w) - static_cast<int>(keys.s);
  const int turn = static_cast<int>(keys.d) - static_cast<int>(keys.a);

  if (translation == 0 && turn == 0) {
    return {};
  }
  if (translation == 0) {
    return {static_cast<double>(turn), static_cast<double>(-turn)};
  }
  if (turn == 0) {
    return {static_cast<double>(translation), static_cast<double>(translation)};
  }

  const double direction = static_cast<double>(translation);
  if (turn > 0) {
    return {direction, direction * inner_ratio};
  }
  return {direction * inner_ratio, direction};
}

struct RampConfig {
  double acceleration_per_second{};
  double deceleration_per_second{};
};

class RampState {
 public:
  explicit RampState(RampConfig config, WheelCommand initial = {})
      : config_(config), current_(initial) {
    if (!std::isfinite(config_.acceleration_per_second) ||
        !std::isfinite(config_.deceleration_per_second) ||
        config_.acceleration_per_second <= 0.0 ||
        config_.deceleration_per_second <= 0.0) {
      throw std::invalid_argument("ramp rates must be positive");
    }
  }

  WheelCommand update(WheelCommand target, double elapsed_seconds) {
    if (!std::isfinite(elapsed_seconds) || elapsed_seconds <= 0.0 ||
        !std::isfinite(target.left) || !std::isfinite(target.right)) {
      throw std::invalid_argument("ramp elapsed time must be positive");
    }
    current_.left = updateWheel(current_.left, target.left, elapsed_seconds);
    current_.right = updateWheel(current_.right, target.right, elapsed_seconds);
    return current_;
  }

  WheelCommand emergencyStop() {
    current_ = {};
    return current_;
  }

  WheelCommand current() const { return current_; }

 private:
  static double moveToward(double current, double target, double maximum_delta) {
    if (current < target) {
      return std::min(current + maximum_delta, target);
    }
    return std::max(current - maximum_delta, target);
  }

  double updateWheel(double current, double target, double elapsed_seconds) const {
    if (current * target < 0.0) {
      return moveToward(current, 0.0,
                        config_.deceleration_per_second * elapsed_seconds);
    }
    const bool increasing_magnitude = std::fabs(target) > std::fabs(current);
    const double rate = increasing_magnitude
                            ? config_.acceleration_per_second
                            : config_.deceleration_per_second;
    return moveToward(current, target, rate * elapsed_seconds);
  }

  RampConfig config_;
  WheelCommand current_;
};

#pragma once

#include "drive_logic.hpp"

#include <algorithm>
#include <cmath>
#include <cstdint>
#include <limits>
#include <stdexcept>

struct MotorControlConfig {
  double acceleration_per_second{};
  double deceleration_per_second{};
  double maximum_command{1.0};
  std::int64_t watchdog_ms{100};
  bool invert_left{};
  bool invert_right{};
};

struct MotorTickResult {
  WheelCommand applied;
  double requested_linear_x{};
  double requested_angular_z{};
  std::uint32_t command_age_ms{std::numeric_limits<std::uint32_t>::max()};
  bool watchdog_stopped{true};
};

class MotorControlCore {
 public:
  explicit MotorControlCore(MotorControlConfig config)
      : config_(config),
        ramp_({config.acceleration_per_second,
               config.deceleration_per_second}) {
    if (!std::isfinite(config_.maximum_command) ||
        config_.maximum_command <= 0.0 || config_.maximum_command > 1.0) {
      throw std::invalid_argument("maximum command must be in (0, 1]");
    }
    if (config_.watchdog_ms < 40) {
      throw std::invalid_argument("watchdog must be at least 40 ms");
    }
  }

  void acceptCommand(double linear_x, double angular_z,
                     std::int64_t now_ms) {
    if (!std::isfinite(linear_x) || !std::isfinite(angular_z)) {
      throw std::invalid_argument("motion command must be finite");
    }
    if (now_ms < 0) {
      throw std::invalid_argument("command time cannot be negative");
    }
    requested_linear_x_ = linear_x;
    requested_angular_z_ = angular_z;
    last_command_ms_ = now_ms;
    has_command_ = true;
  }

  MotorTickResult tick(std::int64_t now_ms, double elapsed_seconds) {
    if (!std::isfinite(elapsed_seconds) || elapsed_seconds <= 0.0) {
      throw std::invalid_argument("elapsed time must be positive");
    }
    if (now_ms < 0 || (has_command_ && now_ms < last_command_ms_)) {
      throw std::invalid_argument("control clock moved backward");
    }

    MotorTickResult result;
    result.requested_linear_x = requested_linear_x_;
    result.requested_angular_z = requested_angular_z_;
    if (!has_command_) {
      result.applied = ramp_.emergencyStop();
      return result;
    }

    const std::int64_t age = now_ms - last_command_ms_;
    result.command_age_ms = static_cast<std::uint32_t>(std::min<std::int64_t>(
        age, std::numeric_limits<std::uint32_t>::max()));
    if (age >= config_.watchdog_ms) {
      result.applied = ramp_.emergencyStop();
      return result;
    }

    double left = requested_linear_x_ - requested_angular_z_;
    double right = requested_linear_x_ + requested_angular_z_;
    const double divisor =
        std::max({1.0, std::fabs(left), std::fabs(right)});
    left = left / divisor * config_.maximum_command;
    right = right / divisor * config_.maximum_command;
    if (config_.invert_left) {
      left = -left;
    }
    if (config_.invert_right) {
      right = -right;
    }

    result.applied = ramp_.update({left, right}, elapsed_seconds);
    result.watchdog_stopped = false;
    return result;
  }

  WheelCommand emergencyStop() { return ramp_.emergencyStop(); }

 private:
  MotorControlConfig config_;
  RampState ramp_;
  bool has_command_{};
  std::int64_t last_command_ms_{};
  double requested_linear_x_{};
  double requested_angular_z_{};
};

#pragma once

#include <array>
#include <charconv>
#include <cmath>
#include <cstdint>
#include <iomanip>
#include <locale>
#include <sstream>
#include <stdexcept>
#include <string>
#include <string_view>
#include <system_error>

namespace remote_mvp {

struct RemoteCommand {
  std::uint64_t session_id;
  std::uint64_t sequence;
  double linear_x;
  double angular_z;
};

namespace detail {

inline bool isDecimalFloatingPoint(std::string_view field) {
  std::size_t index = 0;
  if (index < field.size() && (field[index] == '+' || field[index] == '-')) {
    ++index;
  }

  bool integer_digits = false;
  while (index < field.size() && field[index] >= '0' &&
         field[index] <= '9') {
    integer_digits = true;
    ++index;
  }

  bool fraction_digits = false;
  if (index < field.size() && field[index] == '.') {
    ++index;
    while (index < field.size() && field[index] >= '0' &&
           field[index] <= '9') {
      fraction_digits = true;
      ++index;
    }
  }
  if (!integer_digits && !fraction_digits) {
    return false;
  }

  if (index < field.size() && (field[index] == 'e' || field[index] == 'E')) {
    ++index;
    if (index < field.size() && (field[index] == '+' || field[index] == '-')) {
      ++index;
    }
    const auto exponent_start = index;
    while (index < field.size() && field[index] >= '0' &&
           field[index] <= '9') {
      ++index;
    }
    if (index == exponent_start) {
      return false;
    }
  }
  return index == field.size();
}

inline void validateCommandValues(const RemoteCommand& command) {
  if (!std::isfinite(command.linear_x) || !std::isfinite(command.angular_z)) {
    throw std::invalid_argument("remote command values must be finite");
  }
  if (command.linear_x < -1.0 || command.linear_x > 1.0 ||
      command.angular_z < -1.0 || command.angular_z > 1.0) {
    throw std::invalid_argument("remote command values must be in [-1, 1]");
  }
}

inline std::uint64_t parseUnsigned(std::string_view field,
                                   const char* field_name) {
  if (field.empty()) {
    throw std::invalid_argument(std::string("missing ") + field_name);
  }
  std::uint64_t value = 0;
  const auto result = std::from_chars(field.data(), field.data() + field.size(),
                                      value, 10);
  if (result.ec != std::errc{} || result.ptr != field.data() + field.size()) {
    throw std::invalid_argument(std::string("invalid ") + field_name);
  }
  return value;
}

inline double parseDouble(std::string_view field, const char* field_name) {
  if (field.empty() || !isDecimalFloatingPoint(field)) {
    throw std::invalid_argument(std::string("missing ") + field_name);
  }

  std::istringstream stream{std::string(field)};
  stream.imbue(std::locale::classic());
  double value = 0.0;
  stream >> value;
  if (stream.fail() || stream.peek() != std::char_traits<char>::eof() ||
      !std::isfinite(value)) {
    throw std::invalid_argument(std::string("invalid ") + field_name);
  }
  return value;
}

}  // namespace detail

inline std::string encodeRemoteCommand(const RemoteCommand& command) {
  detail::validateCommandValues(command);

  std::ostringstream stream;
  stream.imbue(std::locale::classic());
  stream << "ROVER1 " << command.session_id << ' ' << command.sequence << ' '
         << std::setprecision(17) << command.linear_x << ' ' << command.angular_z
         << '\n';
  const auto payload = stream.str();
  if (payload.size() > 128) {
    throw std::invalid_argument("encoded remote command exceeds 128 bytes");
  }
  return payload;
}

inline RemoteCommand parseRemoteCommand(std::string_view payload) {
  if (payload.size() > 128) {
    throw std::invalid_argument("remote command exceeds 128 bytes");
  }
  if (payload.size() < 2 || payload.back() != '\n') {
    throw std::invalid_argument("remote command must end with one newline");
  }

  const auto body = payload.substr(0, payload.size() - 1);
  std::array<std::string_view, 5> fields{};
  std::size_t start = 0;
  for (std::size_t index = 0; index < fields.size() - 1; ++index) {
    const auto separator = body.find(' ', start);
    if (separator == std::string_view::npos) {
      throw std::invalid_argument("remote command has missing fields");
    }
    fields[index] = body.substr(start, separator - start);
    start = separator + 1;
  }
  fields.back() = body.substr(start);
  if (fields[0] != "ROVER1" || fields[0].empty() || fields[1].empty() ||
      fields[2].empty() || fields[3].empty() || fields[4].empty() ||
      fields[4].find(' ') != std::string_view::npos) {
    throw std::invalid_argument("malformed remote command");
  }

  RemoteCommand command{
      detail::parseUnsigned(fields[1], "session id"),
      detail::parseUnsigned(fields[2], "sequence"),
      detail::parseDouble(fields[3], "linear command"),
      detail::parseDouble(fields[4], "angular command")};
  detail::validateCommandValues(command);
  return command;
}

class RemoteSequenceGate {
 public:
  bool accept(const RemoteCommand& command) {
    if (!has_last_ || command.session_id != session_id_) {
      has_last_ = true;
      session_id_ = command.session_id;
      sequence_ = command.sequence;
      return true;
    }
    if (command.sequence <= sequence_) {
      return false;
    }
    sequence_ = command.sequence;
    return true;
  }

 private:
  bool has_last_ = false;
  std::uint64_t session_id_ = 0;
  std::uint64_t sequence_ = 0;
};

}  // namespace remote_mvp

#include "motor_teleop_v2/motor_control_core.hpp"
#include "remote_mvp/remote_command.hpp"
#include <iomanip>
#include <iostream>
#include <string>

// Deterministic timestamps, no sockets, serial port, timers, or motor drivers.
int main() {
    MotorControlCore core({0.75, 1.5, 1.0, 100, false, false});
    remote_mvp::RemoteSequenceGate sequence;
    std::cout << "time_ms,accepted,left,right,command_age_ms,watchdog_stopped\n";
    std::cout << std::fixed << std::setprecision(3);
    for (int now = 0; now <= 200; now += 20) {
        std::string packet;
        if (now == 0 || now == 20) packet = "ROVER1 1 0 0.8 0\n";
        if (now == 40) packet = "ROVER1 1 1 0.8 0.2\n";
        if (now == 60) packet = "ROVER1 1 2 nan 0\n";
        if (now == 180) packet = "ROVER1 1 2 0.4 0\n";
        bool accepted = false;
        if (!packet.empty()) {
            try {
                const auto command = remote_mvp::parseRemoteCommand(packet);
                if (sequence.accept(command)) {
                    core.acceptCommand(command.linear_x, command.angular_z, now);
                    accepted = true;
                }
            } catch (const std::invalid_argument&) {
                // Malformed input cannot refresh the watchdog.
            }
        }
        const auto result = core.tick(now, 0.020);
        std::cout << now << ',' << accepted << ',' << result.applied.left << ','
                  << result.applied.right << ',' << result.command_age_ms << ','
                  << result.watchdog_stopped << '\n';
    }
}

#include "drive_logic.hpp"

#include <cmath>
#include <cstdlib>
#include <iostream>
#include <limits>
#include <stdexcept>
#include <string>

namespace {

int failures = 0;

void expectNear(double actual, double expected, const std::string& message) {
  if (std::fabs(actual - expected) > 1e-9) {
    std::cerr << "FAIL: " << message << ": expected " << expected
              << ", got " << actual << '\n';
    ++failures;
  }
}

void expectCommand(const KeyState& keys, double left, double right,
                   const std::string& message) {
  const auto command = mixKeys(keys, 0.20);
  expectNear(command.left, left, message + " left");
  expectNear(command.right, right, message + " right");
}

}  // namespace

int main() {
  expectCommand({}, 0.0, 0.0, "neutral stops");
  expectCommand({true, false, false, false}, 1.0, 1.0, "W moves forward");
  expectCommand({false, false, true, false}, -1.0, -1.0, "S moves backward");
  expectCommand({false, true, false, false}, -1.0, 1.0, "A pivots left");
  expectCommand({false, false, false, true}, 1.0, -1.0, "D pivots right");
  expectCommand({true, true, false, false}, 0.2, 1.0, "W+A curves left");
  expectCommand({true, false, false, true}, 1.0, 0.2, "W+D curves right");
  expectCommand({false, true, true, false}, -0.2, -1.0, "S+A curves left");
  expectCommand({false, false, true, true}, -1.0, -0.2, "S+D curves right");
  expectCommand({true, false, true, false}, 0.0, 0.0, "W+S cancels");
  expectCommand({false, true, false, true}, 0.0, 0.0, "A+D cancels");
  expectCommand({true, false, true, true}, 1.0, -1.0,
                "W+S+D reduces to D");
  expectCommand({true, true, true, false}, -1.0, 1.0,
                "W+A+S reduces to A");
  expectCommand({true, true, false, true}, 1.0, 1.0,
                "W+A+D reduces to W");
  expectCommand({false, true, true, true}, -1.0, -1.0,
                "A+S+D reduces to S");
  expectCommand({true, true, true, true}, 0.0, 0.0, "all keys cancel");

  bool rejected_ratio = false;
  try {
    (void)mixKeys({}, 1.01);
  } catch (const std::invalid_argument&) {
    rejected_ratio = true;
  }
  if (!rejected_ratio) {
    std::cerr << "FAIL: inner-wheel ratios above one are rejected\n";
    ++failures;
  }
  bool rejected_nan_ratio = false;
  try {
    (void)mixKeys({}, std::numeric_limits<double>::quiet_NaN());
  } catch (const std::invalid_argument&) {
    rejected_nan_ratio = true;
  }
  if (!rejected_nan_ratio) {
    std::cerr << "FAIL: non-finite inner-wheel ratio is rejected\n";
    ++failures;
  }

  RampState ramp({1.0, 2.0});
  auto ramped = ramp.update({1.0, -1.0}, 0.1);
  expectNear(ramped.left, 0.1, "left accelerates forward");
  expectNear(ramped.right, -0.1, "right accelerates backward");

  ramp = RampState({1.0, 2.0}, {0.5, -0.5});
  ramped = ramp.update({0.0, 0.0}, 0.1);
  expectNear(ramped.left, 0.3, "left decelerates toward zero");
  expectNear(ramped.right, -0.3, "right decelerates toward zero");

  ramp = RampState({1.0, 2.0}, {0.5, 0.1});
  ramped = ramp.update({-1.0, -1.0}, 0.1);
  expectNear(ramped.left, 0.3, "reversal first decelerates");
  expectNear(ramped.right, 0.0, "reversal clamps at zero");
  ramped = ramp.update({-1.0, -1.0}, 0.1);
  expectNear(ramped.left, 0.1, "continued reversal keeps decelerating");
  expectNear(ramped.right, -0.1, "reversal accelerates after zero");

  ramp = RampState({1.0, 2.0}, {0.05, -0.05});
  ramped = ramp.update({0.0, 0.0}, 0.1);
  expectNear(ramped.left, 0.0, "deceleration cannot overshoot zero");
  expectNear(ramped.right, 0.0, "negative deceleration cannot overshoot zero");

  ramp = RampState({1.0, 2.0}, {0.8, -0.7});
  ramped = ramp.emergencyStop();

#include "remote_mvp/remote_command.hpp"

#include <cmath>
#include <cstdlib>
#include <functional>
#include <iostream>
#include <limits>
#include <stdexcept>
#include <string>

namespace {

int failures = 0;

void expect(bool condition, const std::string& message) {
  if (!condition) {
    std::cerr << "FAIL: " << message << '\n';
    ++failures;
  }
}

bool rejects(const std::function<void()>& operation) {
  try {
    operation();
    return false;
  } catch (const std::invalid_argument&) {
    return true;
  }
}

}  // namespace

int main() {
  using remote_mvp::RemoteCommand;
  using remote_mvp::RemoteSequenceGate;
  using remote_mvp::encodeRemoteCommand;
  using remote_mvp::parseRemoteCommand;

  const RemoteCommand input{44, 7, 0.5, -0.25};
  const auto encoded = encodeRemoteCommand(input);
  expect(encoded.rfind("ROVER1 ", 0) == 0, "exact ROVER1 prefix");
  expect(!encoded.empty() && encoded.back() == '\n',
         "encoded command ends with one newline");
  const auto output = parseRemoteCommand(encoded);
  expect(output.session_id == 44 && output.sequence == 7,
         "round-trip identity");
  expect(output.linear_x == input.linear_x && output.angular_z == input.angular_z,
         "round-trip command values");
  expect(encodeRemoteCommand({1, 2, 0.1, -0.2}) ==
             "ROVER1 1 2 0.10000000000000001 -0.20000000000000001\n",
         "classic-locale 17-digit encoding");

  const RemoteCommand boundary{0, std::numeric_limits<std::uint64_t>::max(),
                               -1.0, 1.0};
  const auto boundary_output = parseRemoteCommand(encodeRemoteCommand(boundary));
  expect(boundary_output.session_id == boundary.session_id &&
             boundary_output.sequence == boundary.sequence &&
             boundary_output.linear_x == boundary.linear_x &&
             boundary_output.angular_z == boundary.angular_z,
         "negative and boundary values round-trip");

  expect(rejects([] { parseRemoteCommand(std::string(129, 'x')); }),
         "payloads over 128 bytes are rejected");
  expect(rejects([] { parseRemoteCommand("ROVER2 1 2 0 0\n"); }),
         "wrong magic is rejected");
  expect(rejects([] { parseRemoteCommand("ROVER1 1 2 0\n"); }),
         "missing field is rejected");
  expect(rejects([] { parseRemoteCommand("ROVER1 1 2 0 0 9\n"); }),
         "extra field is rejected");
  expect(rejects([] { parseRemoteCommand("ROVER1 1 2 nan 0\n"); }),
         "NaN is rejected");
  expect(rejects([] { parseRemoteCommand("ROVER1 1 2 0 inf\n"); }),
         "infinity is rejected");
  expect(rejects([] { parseRemoteCommand("ROVER1 1 2 1.0000001 0\n"); }),
         "linear value above range is rejected");
  expect(rejects([] { parseRemoteCommand("ROVER1 1 2 0 -1.0000001\n"); }),
         "angular value below range is rejected");
  expect(rejects([] { parseRemoteCommand("ROVER1 1 2 0 0\ntrailing"); }),
         "trailing data is rejected");
  expect(rejects([] { parseRemoteCommand("ROVER1 1 2 0 0"); }),
         "missing terminating newline is rejected");
  expect(rejects([] { parseRemoteCommand("ROVER1 1 2 0 0\n\n"); }),
         "extra newline is rejected");
  expect(rejects([] { parseRemoteCommand("ROVER1 1 2 oops 0\n"); }),
         "malformed numeric value is rejected");
  expect(rejects([] { parseRemoteCommand("ROVER1 1 2 0x0 0\n"); }),
         "hexadecimal numeric value is rejected");
  expect(rejects([] { parseRemoteCommand("ROVER1 -1 2 0 0\n"); }),
         "negative session is rejected");
  expect(rejects([] {
           encodeRemoteCommand({0, 0, std::numeric_limits<double>::quiet_NaN(), 0});
         }),
         "encoding NaN is rejected");
  expect(rejects([] {
           encodeRemoteCommand({0, 0, 0, std::numeric_limits<double>::infinity()});
         }),
         "encoding infinity is rejected");
  expect(rejects([] { encodeRemoteCommand({0, 0, 1.01, 0}); }),
         "encoding out-of-range value is rejected");

  RemoteSequenceGate gate;
  expect(gate.accept({44, 7, 0, 0}), "first sequence is accepted");
  expect(!gate.accept({44, 7, 0, 0}), "duplicate sequence is rejected");
  expect(!gate.accept({44, 6, 0, 0}), "older sequence is rejected");
  expect(gate.accept({45, 0, 0, 0}), "new session is accepted");
  expect(gate.accept({45, 1, 0, 0}), "newer sequence is accepted");

  if (failures != 0) {
    std::cerr << failures << " assertion(s) failed\n";
    return EXIT_FAILURE;
  }
  std::cout << "remote command protocol tests passed\n";
  return EXIT_SUCCESS;
}
