#include "Matrix.h"
#include <cstdlib> // For rand() and RAND_MAX

Matrix::Matrix()
{
	rows = 0;
    col = 0;
}

Matrix::Matrix(vector<vector<double>> d)
{
	if (d.empty()) {
        rows = 0;
        col = 0;
        data = d; 
    } else {
        rows = d.size(); 
        col = d[0].size();  
        data = d;
    }
}

Matrix::Matrix(int r, int c, bool isRandom)
{
	rows = r;
	col = c;

	data.resize(rows);
	for(int i = 0; i < rows; ++i) {
		data[i].resize(col);
	}
	if (isRandom) {
		for(int i=0; i<r; ++i){
			for(int j=0; j < c; ++j) {
				data[i][j]=(double)rand() / RAND_MAX;
			}
		}
	}
}

int Matrix::getRows() const { return rows; }
int Matrix::getCol() const { return col; }
vector<vector<double>> Matrix::getData() const { return data; }

void Matrix::print_matrix() const{
	for (int i = 0; i < rows; ++i){
		for (int j = 0; j < col; ++j){
			std::cout << std::fixed << std::setprecision(4) << std::setw(10) << data[i][j] << " ";
		}
		std::cout << endl;
	}
}

Matrix Matrix::operator*(const Matrix& mat) const
{
	if(this->col != mat.getRows()){
		std::cerr<<"Error: Matrix A columm size must be equal to Matrix B rows for multiplication"<<endl;
		return Matrix();
	}
	int result_rows = this->rows;
	int result_col = mat.getCol();
	vector<vector<double>> result(result_rows, vector<double>(result_col, 0.0));

	for (int i = 0; i < result_rows; ++i) {
		for (int j = 0; j < result_col; ++j) {
			double sum = 0.0;
			for (int k = 0; k < this->col; ++k){
				sum += this->data[i][k]*mat.data[k][j]; 
			}
		result[i][j]=sum;
		}
	}
	return Matrix(result);
}

Matrix Matrix::operator+(const Matrix& mat) const
{
	if (this->rows != mat.getRows() || this->col != mat.getCol()){
		std::cerr<< "Error: Matrices must have the same dimensions for addition"<<endl;
		return Matrix();
	}

	vector<vector<double>> result(rows, vector<double>(col));

	for(int i=0; i<rows; ++i) {
		for(int j = 0; j<col; ++j){
			result[i][j] = this->data[i][j] + mat.data[i][j];
		}
	}

	return Matrix(result);
}

Matrix Matrix::operator-(const Matrix& mat) const
{
	if (this->rows != mat.getRows() || this->col != mat.getCol()){
		std::cerr<< "Error: Matrices must have the same dimensions for substraction"<<endl;
		return Matrix();
	}

	vector<vector<double>> result(rows, vector<double>(col));

	for(int i=0; i<rows; ++i) {
		for(int j = 0; j<col; ++j){
			result[i][j] = this->data[i][j] - mat.data[i][j];
		}
	}

	return Matrix(result);
}

Matrix Matrix::operator*(double num) const
{
	vector<vector<double>> result(rows, vector<double>(col));
	for (int i = 0; i < rows; ++i) {
		for(int j = 0; j < col; ++j) {
			result[i][j] = this->data[i][j] * num;
		}
	}
	return Matrix(result);
}

Matrix Matrix::transpose() const
{
	vector<vector<double>> result_data(col, vector<double>(rows));
	for (int i = 0; i < rows; ++i) {
		for (int j = 0; j < col; ++j) {
			result_data[j][i] = this->data[i][j];
		}
	}
	return Matrix(result_data);
}

Matrix Matrix::element_multiply(const Matrix& mat) const
{
	if(this->rows != mat.getRows()||this->col != mat.getCol()) {
		std::cerr << "Error: Matrices must have the same dimensions for element-wise multiplication." << endl;
		exit(1);
	}
	vector<vector<double>> result_data(rows, vector<double>(col));
	for(int i = 0; i < rows; ++i){
		for(int j = 0; j < col; ++j){
			result_data[i][j] = this->data[i][j] * mat.data[i][j];
		}
	}
	return Matrix(result_data);
}

using namespace std;


//A layer refers to the neurons in a single layer
//Each will have their own wieghts and biases

Layer::Layer(int input_size, int output_size, ActivationType act_type) ////////////////
{
	weights = Matrix(output_size, input_size, true);
	bias = Matrix(output_size, 1, true);
	activation = act_type;
}

Layer::Layer() {}


ANN::ANN(vector<int> sizes, vector<ActivationType> activations)
{
	this->layerSizes = sizes;
	for (size_t i = 0; i + 1 < sizes.size(); ++i) {
		layers.push_back(Layer(sizes[i], sizes[i+1], activations[i]));/////////////////
	}
}

Matrix ANN::feedforward(Matrix input)
{
	activations.clear();
	activations.push_back(input); //First activation layer is always the input
	Matrix current_activation = input;

	for (size_t i=0; i<layers.size(); ++i){
		Layer& current_layer = layers[i];
		Matrix z = (current_layer.weights * current_activation) + current_layer.bias;

		switch (current_layer.activation)
        {
            case SIGMOID:
                current_activation = nn_utils::sigmoid(z);
                break;
            case RELU:
                current_activation = nn_utils::relu(z); 
                break;
            case TANH:
                current_activation = nn_utils::tanh(z);
                break;
			// can add more cases, FLEXIBILITY+++
        }
		activations.push_back(current_activation);
	}

	return current_activation;
}

double ANN::update(Matrix target, double training_rate)
{
	Matrix output_activation = activations.back();
	Matrix error_matrix = target - output_activation;

	double absError = 0.0;
	for(int i=0; i<error_matrix.getRows(); ++i) {
		absError += abs(error_matrix.data[i][0]);
	}



	for (int i = layers.size() - 1; i>=0; --i){
		Matrix current_output = activations[i+1];
		Matrix derivative;

		switch(layers[i].activation){
			case SIGMOID:
				derivative = nn_utils::dsigmoid(current_output);
				break;
			case RELU:
				derivative = nn_utils::drelu(current_output);
				break;
			case TANH:
				derivative = nn_utils::dtanh(current_output);
				break;
		}

		Matrix gradient = error_matrix.element_multiply(derivative);
		Matrix scaled_gradient = gradient * training_rate;

		Matrix prev_output = activations[i];
		Matrix prev_output_transposed = prev_output.transpose();
		Matrix delta_weights = scaled_gradient * prev_output_transposed;
		
        // Propagate through the weights used by this forward pass, before updating them.
        Matrix previous_error;
        if (i > 0) previous_error = layers[i].weights.transpose() * gradient;
		layers[i].weights = layers[i].weights + delta_weights;
		layers[i].bias = layers[i].bias + scaled_gradient;
		if(i>0){
            error_matrix = previous_error; 
		}
	}

	return absError;
}


// Utility functions for ANN
Matrix nn_utils::sigmoid(const Matrix& m)
{
	vector<vector<double>> result_data(m.getRows(), vector<double>(m.getCol()));
	for (int i = 0; i < m.getRows(); ++i)
	{
		for (int j = 0; j < m.getCol(); ++j)
		{
			result_data[i][j] = 1.0 / (1.0 + exp(-m.data[i][j]));
		}
	}
	return Matrix(result_data);
}

Matrix nn_utils::relu(const Matrix& m)
{
    Matrix result(m.getRows(), m.getCol(), false);
    for (int i = 0; i < m.getRows(); ++i)
        for (int j = 0; j < m.getCol(); ++j)
            result.data[i][j] = m.data[i][j] > 0 ? m.data[i][j] : 0.0;
    return result;
}

Matrix nn_utils::tanh(const Matrix& m)
{
    Matrix result(m.getRows(), m.getCol(), false);
    for (int i = 0; i < m.getRows(); ++i)
        for (int j = 0; j < m.getCol(); ++j)
            result.data[i][j] = std::tanh(m.data[i][j]);
    return result;
}

Matrix nn_utils::dsigmoid(const Matrix& m)
{
	vector<vector<double>> result_data(m.getRows(), vector<double>(m.getCol()));
	for (int i = 0; i < m.getRows(); ++i)
	{
		for (int j = 0; j < m.getCol(); ++j)
		{
			result_data[i][j] = m.data[i][j] * (1.0 - m.data[i][j]);
		}
	}
	return Matrix(result_data);
}

Matrix nn_utils::drelu(const Matrix& m)
{
    Matrix result(m.getRows(), m.getCol(), false);
    for (int i = 0; i < m.getRows(); ++i)
        for (int j = 0; j < m.getCol(); ++j)
            result.data[i][j] = m.data[i][j] > 0 ? 1.0 : 0.0;
    return result;
}

Matrix nn_utils::dtanh(const Matrix& m)
{
    Matrix result(m.getRows(), m.getCol(), false);
    for (int i = 0; i < m.getRows(); ++i)
        for (int j = 0; j < m.getCol(); ++j)
            result.data[i][j] = 1.0 - m.data[i][j] * m.data[i][j];
    return result;
}

int nn_utils::argmax(const Matrix& m)
{
	if (m.getCol() != 1)
	{
		cout<< "argmax only supports 1-column matrices!";
	}
	double max_val = -99999;
	int max_idx = 0;
	for (int i = 0; i < m.getRows(); i++)
	{
		if (m.data[i][0] > max_val)
		{
			max_val = m.data[i][0];
			max_idx = i;
		}
	}
	return max_idx;
}

# Four-bit neural-network decoder

A small C++ matrix and feedforward neural-network exercise. The demo trains a 4–8–16 network to map each four-bit input to one of sixteen one-hot output classes, using sigmoid activations and backpropagation without a machine-learning framework.

## Build, train and test

Requires Make and a GCC/Clang C++17 compiler. There are no third-party runtime dependencies.

```sh
make
./build/decoder
make test
make clean
```

The demo uses seed 7, 10,000 epochs and learning rate 0.1. It reports predictions for all sixteen inputs and exits nonzero unless all are classified correctly. `make test` stores the training transcript in ignored `build/training.log`.

The update rule optimises half the sum of squared output errors. The progress metric printed as `Avg Loss` is the average per-example sum of absolute errors; it is not the squared-error optimisation objective. Activation derivative helpers take the already activated output. ReLU uses derivative zero at zero.

## Provenance and changes

`Matrix.h/.cpp` and `NeuralNetwork.h/.cpp` carry Ivan Ling's University of Southampton teaching-material headers. Those headers and copyright notices are retained. The recovered decoder application and coursework modifications are associated with Emre Karausta. This example is not listed in the master CV.

Preparation completed the four unfinished ReLU/tanh activation and derivative functions, corrected hidden-layer error propagation to use the forward pass weights, and made the example reproducible and checked across all sixteen inputs. Added numerical gradient and activation tests. Original files were not changed.

No blanket open-source licence is granted for instructor material. Publication/redistribution rights must be confirmed separately.

See [VERIFICATION.md](VERIFICATION.md). Fixed model dimensions are intentional: callers must provide matching matrix, target and layer shapes; this educational API does not validate every malformed shape.

#include <iostream>
#include <vector>
#include <cstdlib>

#include <iomanip>
#include "NeuralNetwork.h" 

using namespace std;


Matrix create_input_matrix(int n)
{
    vector<vector<double>> d(4, vector<double>(1, 0.0));

    d[0][0] = (n >> 3) & 1;
    d[1][0] = (n >> 2) & 1;
    d[2][0] = (n >> 1) & 1;
    d[3][0] = (n >> 0) & 1;
    return Matrix(d);
}

Matrix create_target_matrix(int n)
{
    vector<vector<double>> d(16, vector<double>(1, 0.0));
    if (n >= 0 && n < 16) {
        d[n][0] = 1.0;
    }
    return Matrix(d);
}

int main()
{
    srand(7); // Reproducible training demo.
    cout << "--- 4-Bit Binary Decoder Training ---" << endl << endl;
    vector<int> network_sizes = {4, 8, 16};

    vector<ActivationType> activations = {SIGMOID, SIGMOID};

    ANN decoder(network_sizes, activations);
   
    double training_rate = 0.1;
    int epochs = 10000; 
    cout << "Network: 4-8-16 | Epochs: " << epochs << " | Rate: " << training_rate << endl;

    vector<Matrix> inputs;
    vector<Matrix> targets;

    for (int i = 0; i < 16; i++)
    {
        inputs.push_back(create_input_matrix(i));
        targets.push_back(create_target_matrix(i));
    }
    cout << "Generated 16 input/target pairs." << endl << endl;

    cout << "Starting training..." << endl;
    cout << std::fixed << std::setprecision(8);

    for (int i = 0; i < epochs; i++)
    {
        double total_epoch_loss = 0.0;
        for (int j = 0; j < 16; j++)
        {
            decoder.feedforward(inputs[j]); 
            total_epoch_loss += decoder.update(targets[j], training_rate);
        }
        if ((i + 1) % 100 == 0)
        {
            cout << "Epoch: " << setw(5) << (i + 1) << " | Avg Loss: " << (total_epoch_loss / 16.0) << endl;
        }
    }

    cout << "\n--- Training Complete: Testing ---" << endl;
    cout << std::fixed << std::setprecision(2);

    vector<int> test_values;
    for (int i = 0; i < 16; ++i) test_values.push_back(i);
    int correct = 0;

    for (int val : test_values)
    {
        Matrix test_input = create_input_matrix(val);
        Matrix test_output = decoder.feedforward(test_input);

        cout << "-----------------------------------" << endl;
        cout << "Input: " << val << endl;
        cout << "Correct Output Index: " << val << endl;
        
        int predicted = nn_utils::argmax(test_output);
        correct += predicted == val;
        cout << "Network Output Index: " << predicted << endl;

        cout << "Output Vector: ";
        test_output.transpose().print_matrix();
        cout << endl;
    }
    cout << "Accuracy: " << correct << "/16" << endl;
    return correct == 16 ? 0 : 1;
}

// Test-only visibility lets finite differences inspect the actual ANN parameters.
#include <vector>
#include <iostream>
#include <iomanip>
#include "NeuralNetwork.h"
#include <cassert>
#include <cmath>
#include <cstdlib>
double loss(ANN& net, const Matrix& x, const Matrix& y) {
    Matrix out=net.feedforward(x);
    double value=0;
    for(int i=0;i<out.getRows();++i) value+=0.5*std::pow(out.data[i][0]-y.data[i][0],2);
    return value;
}
int main(int argc,char**argv) {
    if(argc>1 && std::string(argv[1])=="gradient") {
        std::srand(7);
        ANN net({2,3,2},{SIGMOID,SIGMOID});
        Matrix x(std::vector<std::vector<double>>{{0.2},{0.7}}), y(std::vector<std::vector<double>>{{0.1},{0.9}});
        ANN before=net;
        net.feedforward(x); net.update(y,0.5);
        double worst=0;
        for(size_t l=0;l<before.layers.size();++l) {
            for(int kind=0;kind<2;++kind) {
                Matrix& params=kind ? before.layers[l].bias : before.layers[l].weights;
                Matrix& actual=kind ? net.layers[l].bias : net.layers[l].weights;
                for(int r=0;r<params.getRows();++r) for(int c=0;c<params.getCol();++c) {
                    double v=params.data[r][c], eps=1e-5;
                    params.data[r][c]=v+eps; double plus=loss(before,x,y);
                    params.data[r][c]=v-eps; double minus=loss(before,x,y);
                    params.data[r][c]=v;
                    double numeric=(plus-minus)/(2*eps);
                    double analytic=(v-actual.data[r][c])/0.5;
                    worst=std::max(worst,std::abs(numeric-analytic));
                }
            }
        }
        std::cout<<"Gradient max absolute error: "<<worst<<std::endl;
        assert(worst<1e-7);
    } else {
        Matrix m(std::vector<std::vector<double>>{{-2,0,2}});
        Matrix r=nn_utils::relu(m), t=nn_utils::tanh(m), dr=nn_utils::drelu(r), dt=nn_utils::dtanh(t);
        assert(r.data[0][0]==0 && r.data[0][1]==0 && r.data[0][2]==2);
        assert(std::abs(t.data[0][0]+0.9640275800758169)<1e-12);
        assert(dr.data[0][0]==0 && dr.data[0][1]==0 && dr.data[0][2]==1);
        assert(std::abs(dt.data[0][0]-0.07065082485316443)<1e-12 && dt.data[0][1]==1);
        std::cout<<"Activation values and derivatives PASS\n";
    }
}
